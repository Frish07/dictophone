cmake_minimum_required(VERSION 3.20)
project(Dictophone VERSION 1.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Dictophone is a Windows 10/11 application")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dictophone_audio src/audio.cpp)
target_compile_definitions(dictophone_audio PUBLIC UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(dictophone_audio PUBLIC ole32 uuid ksuser)
add_executable(Dictophone WIN32 src/main.cpp src/app.rc)
target_link_libraries(Dictophone PRIVATE dictophone_audio comctl32 shell32 dwmapi gdi32)
target_include_directories(Dictophone PRIVATE src)
if(MSVC)
  target_compile_options(dictophone_audio PRIVATE /W4 /utf-8)
  target_compile_options(Dictophone PRIVATE /W4 /utf-8)
  # app.rc already embeds our DPI/compatibility manifest as resource 1.
  target_link_options(Dictophone PRIVATE /MANIFEST:NO)
  set_property(TARGET dictophone_audio Dictophone PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(Dictophone PRIVATE -mwindows -municode -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(AudioTests tests/audio_tests.cpp)
  target_link_libraries(AudioTests PRIVATE dictophone_audio shell32)
  if(MSVC)
    target_compile_options(AudioTests PRIVATE /W4 /utf-8)
    set_property(TARGET AudioTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_link_options(AudioTests PRIVATE -municode -static)
  endif()
  add_test(NAME audio COMMAND AudioTests)
endif()
